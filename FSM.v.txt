module FSM_Controller (
    input wire clk,
    input wire rst,
    input wire [7:0] hum,
    input wire valid,
    output reg pump_status // 1 for ON, 0 for OFF
);
    always @(posedge clk or posedge rst) begin
        if (rst) 
            pump_status <= 0;
        else if (valid) begin
            if (hum < 40) // Example threshold
                pump_status <= 1;
            else 
                pump_status <= 0;
        end
    end
endmodule
